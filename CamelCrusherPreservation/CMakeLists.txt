cmake_minimum_required(VERSION 3.25)
project(CamelCrusherPreservation VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(CAMEL_BUILD_LEGACY "Build the independent legacy ABI compatibility adapter" OFF)
set(VST3_SDK_ROOT "" CACHE PATH "Path to a recursive VST3 SDK checkout")
add_library(camel_core STATIC src/Dsp.cpp src/State.cpp)
target_include_directories(camel_core PUBLIC src)
set_target_properties(camel_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
 target_compile_options(camel_core PRIVATE /fp:strict)
else()
 target_compile_options(camel_core PRIVATE -ffp-contract=off -fno-fast-math)
endif()

 function(camel_add_native_editor target objc_class)
  add_library(${target} STATIC src/NativeEditor.mm)
  target_include_directories(${target} PUBLIC src)
  target_compile_options(${target} PRIVATE -fobjc-arc)
  target_compile_definitions(${target} PRIVATE CAMEL_EDITOR_CLASS=${objc_class})
  target_link_libraries(${target} PRIVATE "-framework Cocoa")
  set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
 endfunction()
 camel_add_native_editor(camel_native_vst3 CCPVst3EditorView)
 camel_add_native_editor(camel_native_legacy CCPLegacyEditorView)
 camel_add_native_editor(camel_native_au CCPAUEditorView)
endif()
